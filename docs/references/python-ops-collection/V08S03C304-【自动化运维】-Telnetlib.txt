tn = telnetlib.Telnet(‘192.168.111.1’,23,10) #登录设备

tn.read_until() #直到读取到XXX
tn.read_very_eager() #读取全部

Telnet远程登录
import telnetlib #引入模块
user_name = ‘huawei’
pass_word = ‘Huawei@123’
host_ip_list = [‘10.0.0.1’, ‘10.0.0.2’, ‘10.0.0.3’, ‘10.0.0.4’, ‘10.0.0.5’] #定义设备IP列表
cmd_list = [‘sys’,’stelnet server enable’,……] #定义命令列表
for host_ip in host_ip_list :
conf_info = ‘’ #记录Python的操作
tn = telnetlib.Telnet(host_ip,23,10) #登录设备
tn.read_until(b‘Username:’) #等待设备出现输入用户名的提示
tn.write(user_name.encode('utf-8')+b'\n')
tn.read_until(b‘Password:’) #等待设备出现输入密码的提示
tn.write(pass_word.encode('utf-8')+b'\n’)
for cmd in cmd_list: #循环输入配置命令
tn.write(cmd.encode('utf-8')+ b'\n’)
tn.write(b‘abcdefg\n’) #输入设备不识别的命令
conf_info += tn.read_until(b’abedefg’) #读取设备上面出现的‘abcdefg’
tn.close()                                                                                                              #关闭telnet

Telnetlib3
import telnetlib3,asyncio,time
async def login(): # 使用telnet登录设备，reader用于读取数据，writer用于向设备发送数据
reader,writer = await telnetlib3.open_connection('192.168.110.1',23) 
await reader.read(1024) # 读取设备登录信息，最多1024字节
writer.write('huawei' + '\n’) # 发送用户名
await writer.drain() # 确保数据发送完成
await asyncio.sleep(1) # 等待1s，让设备处理数据
writer.write('Huawei@123' + '\n')
await writer.drain()
await asyncio.sleep(1)
writer.write('disp ip int br \n')
await asyncio.sleep(2)
config = await reader.read(65535)      # 读取数据
print(config)           # 打印数据
writer.close()                    # 关闭写入模块连接
await writer.wait_closed()    # 关闭telnet连接
asyncio.run(login()) # main调用异步函数
